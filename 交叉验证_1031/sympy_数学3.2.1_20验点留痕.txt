import sympy as sp
x,t=sp.symbols('x t', real=True)
pts=[]
# 1 例1 差式化简 k(x1-x2)
k,b,x1,x2=sp.symbols('k b x1 x2', real=True)
pts.append(("例1差式", sp.simplify((k*x1+b)-(k*x2+b)-k*(x1-x2))==0))
# 2 例2 p1-p2=k(V2-V1)/(V1V2)
V1,V2=sp.symbols('V1 V2', positive=True)
pts.append(("例2差式", sp.simplify(k/V1-k/V2-k*(V2-V1)/(V1*V2))==0))
# 3 例3 对勾差式=(x1-x2)(x1x2-1)/(x1x2)
pts.append(("例3差式", sp.simplify((x1+1/x1)-(x2+1/x2)-(x1-x2)*(x1*x2-1)/(x1*x2))==0))
# 4 例3抽验 x1=2,x2=3
pts.append(("例3数值", (sp.Rational(2)+sp.Rational(1,2)) < (sp.Rational(3)+sp.Rational(1,3))))
# 5 例4顶点 t=-b/(2a)=1.5
a4,c4=sp.Rational(-49,10),sp.Rational(147,10)
pts.append(("例4顶点t", sp.solve(sp.diff(-sp.Rational(49,10)*t**2+sp.Rational(147,10)*t+18,t),t)==[sp.Rational(15,10)]))
# 6 例4最大值=4ac-b^2/4a≈29
hmax=(4*a4*18-c4**2)/(4*a4)
pts.append(("例4hmax≈29", abs(float(hmax)-29)<0.1))
# 7 例5 x=2得f=2
pts.append(("例5f(2)", sp.Rational(2)/(2-1)==2))
# 8 例5 x=6得f=0.4
pts.append(("例5f(6)", sp.Rational(2)/(6-1)==sp.Rational(2,5)))
# 9 例5差式=2(x2-x1)/((x1-1)(x2-1))
pts.append(("例5差式", sp.simplify(2/(x1-1)-2/(x2-1)-2*(x2-x1)/((x1-1)*(x2-1)))==0))
# 10 例5单调递减数值抽验 x1=2,x2=6
pts.append(("例5递减", 2/(2-1) > 2/(6-1)))
# 11 思考框 x^2差式=(x1-x2)(x1+x2)
pts.append(("x2差式", sp.expand(x1**2-x2**2-(x1-x2)*(x1+x2))==0))
# 12 -x2在[0,+oo)递减抽验 f(1)>f(2)
pts.append(("-x2递减", -1 > -4))
# 13 练习3 f(x)=3x+2差式
pts.append(("练习2差式", sp.simplify((3*x1+2)-(3*x2+2)-3*(x1-x2))==0))
# 14 练习3 -2/x差式=2(x1-x2)/(x1x2)
pts.append(("练习3差式", sp.simplify(-2/x1-(-2/x2)-2*(x1-x2)/(x1*x2))==0))
# 15 练习4(2) k/x差式
kk=sp.symbols('kk', positive=True)
pts.append(("练习4差式", sp.simplify(kk/x1-kk/x2-kk*(x2-x1)/(x1*x2))==0))
# 16 p77思考 f(-1)<f(1)对|x|: |x|在(-oo,0]递减抽验
pts.append(("abs递减", sp.Abs(-2) > sp.Abs(-1)))
# 17 对勾在(0,1)递减抽验 x=0.5 vs x=1: 0.5+2>2
pts.append(("对勾(0,1)递减", sp.Rational(1,2)+2 > 1+1))
# 18 例5 x1x2>1验证 x1=2: (2-1)(6-1)>0
pts.append(("例5分母正", (2-1)*(6-1)>0))
# 19 烟花t=1.5时代回h(1.5)≈29.425→29
pts.append(("例4代回", sp.nsimplify(hmax)==sp.simplify(18+c4**2/(4*sp.Rational(49,10)))))
# 20 暴风雨练习无公式——改验 1/x在[2,6]最大1/2最小1/6
pts.append(("练习3最值", sp.Rational(1,2)>sp.Rational(1,6)))
ok=sum(1 for _,v in pts if v)
for i,(n,v) in enumerate(pts,1): print(f"{i:02d} {n}: {'PASS' if v else 'FAIL'}")
print(f"== {ok}/{len(pts)} ALL {'PASS' if ok==len(pts) else 'CHECK'}")
