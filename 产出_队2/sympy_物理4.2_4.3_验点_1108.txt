# sympy验点·物理4.2（2024甘肃卷T11）+物理4.3/2.3（2024新课标II卷实验）（并行搜索员A·队2席·1108）
from sympy import symbols, Rational, solve
M, F = symbols('M F', positive=True)
# 验点1（4.2·甘肃T11(3)）：外力一定时 a=F/M，a对(1/M)为过原点直线·斜率=F
a = F/M
print('a=F/M 与 1/M 成正比 -> a-(1/M)图线斜率=合外力F（线性·过原点）: 斜率=slope ->', a*M)
# 甲乙两组同装置异槽码：a甲-(1/M甲)斜率 > a乙-(1/M乙)斜率  <=>  F甲>F乙  <=> m槽码甲>m槽码乙
print('判读(3)自洽: 甲图线斜率更大 -> 甲组合外力更大 -> 甲组所用槽码质量比乙组更大')
# 验点2（4.3·新课标II卷）：a=2(h0-h)/T²=2*(0.590-0.100)/0.730²
h0, h, T = Rational(590,1000), Rational(100,1000), Rational(730,1000)
a_meas = 2*(h0-h)/T**2
print('a=2(h0-h)/T^2 =', float(a_meas), 'm/s² -> 与账面1.84吻合:', abs(float(a_meas)-1.84)<0.005)
# 验点3（4.3·理论a'）: a'=(mB-mA)g/(mA+mB)=(0.150-0.100)*9.8/0.250
mA, mB, g = Rational(100,1000), Rational(150,1000), Rational(98,10)
a_theory = (mB-mA)*g/(mA+mB)
print("a'=(mB-mA)g/(mA+mB) =", float(a_theory), 'm/s² -> 与账面1.96吻合:', abs(float(a_theory)-1.96)<0.005)
# 验点4（2.3对位）：小球B自h0降至h为初速零匀加速·位移x=½aT²反算一致性
x = Rational(490,1000)
print('x=½aT² 反算: ½*1.84*0.730² =', 0.5*1.84*0.730**2, '≈0.490m ✓（2.3位移公式直接应用）')
# 验点5（偏差归因）：实测a(1.84)<理论a'(1.96)·相对偏差
print('相对偏差 =', round((1.96-1.84)/1.96*100,1), '% -> 滑轮轴非理想（摩擦）致实测小于理论·判读自洽')
