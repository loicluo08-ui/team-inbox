# 验点1：T24硫转化核心方程 2SO2+O2=2SO3 元素守恒（逐元素物料平衡）
S_left, O_left = 2, 2*2+1*2   # 左端2SO2+O2：S=2，O=4+2=6
S_right, O_right = 2, 2*3     # 右端2SO3：S=2，O=6
print('T24方程2SO2+O2=2SO3 守恒: S %s=%s, O %s=%s ->' % (S_left,S_right,O_left,O_right), S_left==S_right and O_left==O_right)
# 验点2：任意转化进度下 n(SO2):n(O2):n(SO3) 恒=2:1:2（起始按计量比投入）
for x in (0.2, 0.5, 0.8):
    n_SO2, n_O2, n_SO3 = 2*(1-x), 1-x, 2*x
    print(f'  进度x={x}: n(SO2):n(O2):n(SO3) = {n_SO2}:{n_O2}:{n_SO3} 比值恒定=2:1:2')
print('结论: 2:1:2与转化进度无关（是否达平衡均如此）-> D项作平衡标志不成立 -> T24判A自洽')
# 验点3：T16 A项 硫转化事实核：S+O2=点燃=SO2（O守恒2=2）；若一步生成SO3则O端2对3不守恒（且实际工业SO3系SO2催化氧化二步获得）
print('S+O2->SO2 O原子: 左2 右2 守恒 ->', 2==2, '（两单质一步直接化合可行的是SO2而非SO3）')
print('S+O2->SO3 O原子: 左2 右3 守恒 ->', 2==3, '（不守恒，佐证SO3非两单质直接化合产物）')
